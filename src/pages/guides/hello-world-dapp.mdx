"use client";
import { CardanoWallet, useWallet } from "@meshsdk/react";
import { useEffect, useState } from "react";

export default function Home() {
  const { connected, wallet } = useWallet();
  const [balance, setBalance] = useState("0");
  const [loading, setLoading] = useState(false);

  const getBalance = async () => {
    if (wallet) {

```jsx
"use client";
import { CardanoWallet, useWallet } from "@meshsdk/react";
import { useEffect, useState } from "react";

export default function Home() {
  const { connected, wallet } = useWallet();
  const [balance, setBalance] = useState("0");
  const [loading, setLoading] = useState(false);
  const [sending, setSending] = useState(false);
  const [txHash, setTxHash] = useState("");
  const [error, setError] = useState("");

  const getBalance = async () => {
    if (wallet) {
      setLoading(true);
      try {
        const balance = await wallet.getBalance();
        const adaBalance = parseInt(balance[0].quantity) / 1000000;
        setBalance(adaBalance.toFixed(2));
      } catch (error) {
        console.error("Error getting balance:", error);
        setBalance("Error loading balance");
      }
      setLoading(false);
    }
  };

  const sendTransaction = async () => {
    if (!wallet) return;

    setSending(true);
    setError("");
    setTxHash("");

    try {
      const addresses = await wallet.getUsedAddresses();
      const address = addresses[0];

      const tx = {
        recipients: [
          {
            address: address,
            amount: "1000000", // 1 ADA in lovelace
          },
        ],
        metadata: {
          1337: "Hello World from Mnemos!",
        },
      };

      const unsignedTx = await wallet.buildTx(tx);
      const signedTx = await wallet.signTx(unsignedTx);
      const txHash = await wallet.submitTx(signedTx);

      setTxHash(txHash);

      // Refresh balance after transaction
      setTimeout(() => {
        getBalance();
      }, 3000);
    } catch (error) {
      console.error("Transaction error:", error);
      setError(
        "Transaction failed. Make sure you have enough ADA for the transaction fee (about 0.17 ADA)."
      );
    }

    setSending(false);
  };

  useEffect(() => {
    if (connected) {
      getBalance();
    }
  }, [connected]);

  return (
    <div className="min-h-screen bg-gray-900 text-white p-8">
      <div className="max-w-2xl mx-auto">
        <h1 className="text-4xl font-bold text-center mb-8 bg-gradient-to-r from-sky-400 to-emerald-400 bg-clip-text text-transparent">
          🚀 My First Cardano DApp
        </h1>

        <div className="bg-gray-800 rounded-2xl p-8 mb-6 border border-gray-700">
          <h2 className="text-2xl font-semibold mb-4 text-emerald-400">
            Step 1: Connect Your Wallet
          </h2>
          <p className="text-gray-300 mb-6">
            This connects your Cardano wallet to your app. It's like logging in,
            but for blockchain!
          </p>
          <CardanoWallet />
        </div>

        {connected && (
          <>
            <div className="bg-gradient-to-r from-emerald-900/50 to-green-900/50 border border-emerald-700 rounded-2xl p-8 mb-6">
              <div className="text-center mb-6">
                <div className="text-6xl mb-4">🎉</div>
                <h3 className="text-2xl font-bold text-emerald-400 mb-2">
                  Wallet Connected!
                </h3>
                <p className="text-emerald-200">
                  You're now connected to the Cardano blockchain!
                </p>
              </div>

              <div className="bg-gray-800/50 rounded-xl p-6 text-center">
                <h4 className="text-lg font-semibold mb-2 text-gray-300">
                  Your ADA Balance
                </h4>
                <div className="text-4xl font-bold text-emerald-400 mb-2">
                  {loading ? "..." : `${balance} ADA`}
                </div>
                <p className="text-gray-400 text-sm">
                  {loading
                    ? "Loading your balance..."
                    : "This is your current ADA balance!"}
                </p>
              </div>
            </div>

            <div className="bg-gradient-to-r from-purple-900/50 to-pink-900/50 border border-purple-700 rounded-2xl p-8">
              <div className="text-center mb-6">
                <div className="text-6xl mb-4">🚀</div>
                <h3 className="text-2xl font-bold text-purple-400 mb-2">
                  Send Your First Transaction
                </h3>
                <p className="text-purple-200 mb-4">
                  This will send 1 ADA to yourself with a "Hello World" message!
                </p>
                <p className="text-sm text-gray-400">
                  Don't worry - you'll get your ADA back (minus a small network
                  fee)
                </p>
              </div>

              <div className="text-center">
                <button
                  onClick={sendTransaction}
                  disabled={sending || parseFloat(balance) < 1.2}
                  className={`px-8 py-4 rounded-xl font-bold text-lg transition-all duration-300 ${
                    sending || parseFloat(balance) < 1.2
                      ? "bg-gray-600 cursor-not-allowed"
                      : "bg-gradient-to-r from-purple-600 to-pink-600 hover:from-purple-700 hover:to-pink-700 hover:scale-105"
                  }`}
                >
                  {sending ? "Sending Transaction..." : "Send Hello World! 🌍"}
                </button>

                {parseFloat(balance) < 1.2 && (
                  <p className="text-red-400 text-sm mt-2">
                    You need at least 1.2 ADA to send this transaction (1 ADA +
                    network fee)
                  </p>
                )}
              </div>
            </div>

            {txHash && (
              <div className="bg-gradient-to-r from-green-900/50 to-emerald-900/50 border border-green-700 rounded-2xl p-8 mt-6 text-center">
                <div className="text-6xl mb-4">🎊</div>
                <h3 className="text-2xl font-bold text-green-400 mb-4">
                  Transaction Sent Successfully!
                </h3>
                <p className="text-green-200 mb-4">
                  Congratulations! You just sent your first transaction on the
                  Cardano blockchain!
                </p>
                <div className="bg-gray-800/50 rounded-xl p-4 mb-4">
                  <p className="text-sm text-gray-400 mb-2">
                    Transaction Hash:
                  </p>
                  <p className="text-green-400 font-mono text-sm break-all">
                    {txHash}
                  </p>
                </div>
                <p className="text-sm text-gray-400">
                  This transaction is now permanently recorded on the Cardano
                  blockchain! 🌟
                </p>
              </div>
            )}

            {error && (
              <div className="bg-red-900/50 border border-red-700 rounded-2xl p-6 mt-6 text-center">
                <div className="text-4xl mb-2">⚠️</div>
                <h3 className="text-xl font-bold text-red-400 mb-2">
                  Transaction Failed
                </h3>
                <p className="text-red-200 text-sm">{error}</p>
              </div>
            )}
          </>
        )}
      </div>
    </div>
  );
}
```

### "Wallet not connected"

- Refresh the page and reconnect your wallet
- Make sure your wallet is unlocked

- Check your internet connection
- Try again in a few minutes
